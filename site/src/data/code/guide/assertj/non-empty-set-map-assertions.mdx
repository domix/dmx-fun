```java
import static dmx.fun.assertj.DmxFunAssertions.assertThat;

import org.assertj.core.api.Assertions;

NonEmptySet<String> tags = NonEmptySet.of("java", Set.of("fp", "types"));
NonEmptyMap<String, Integer> scores = NonEmptyMap.of("ana", 10, Map.of("bo", 7));

// Sets: head, size, membership, exact contents in any order
assertThat(tags)
    .hasHead("java")
    .hasSize(3)
    .contains("fp")
    .containsExactlyInAnyOrder("types", "fp", "java");

// Maps: size, keys, entries, and a value checked in place with plain AssertJ
assertThat(scores)
    .hasSize(2)
    .containsKey("bo")
    .containsEntry("ana", 10)
    .hasValueSatisfying("bo", v -> Assertions.assertThat(v).isLessThan(10));
```
